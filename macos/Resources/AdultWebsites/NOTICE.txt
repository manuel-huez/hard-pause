Reproduce domains.txt.deflate from the upstream bytes with Python zlib 1.2.12:
python3 - <<'PY'
from pathlib import Path
import zlib
d = Path("domains.txt").read_bytes()
c = zlib.compressobj(9, zlib.DEFLATED, -zlib.MAX_WBITS, 8, zlib.Z_DEFAULT_STRATEGY)
Path("domains.txt.deflate").write_bytes(c.compress(d) + c.flush(zlib.Z_FINISH))
